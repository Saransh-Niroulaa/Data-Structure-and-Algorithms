#include <iostream>
using namespace std;
class Queue {
    struct Node {
        int data;
        Node* next;
    };
    Node* front = NULL;
    Node* rear = NULL;
public:
    void enqueue(int x) {
        Node* newNode = new Node;
        newNode->data = x;
        newNode->next = NULL;

        if (rear == NULL) {
            front = rear = newNode;
            return;
        }

        rear->next = newNode;
        rear = newNode;
    }
    void dequeue() {
        if (front == NULL) {
            cout << "Queue Underflow\n";
            return;
        }
        Node* temp = front;
        front = front->next;
        delete temp;
        if (front == NULL)
            rear = NULL;
    }
    int peek() {
        return front->data;
    }
    bool empty() {
        return front == NULL;
    }
};
