#include <queue>
using namespace std;

class Stack {
    queue<int> q;

public:
    void push(int x) {
        q.push(x);
        int n = q.size();
        for (int i = 1; i < n; i++) {
            q.push(q.front());
            q.pop();
        }
    }
    void pop() {
        if (!q.empty())
            q.pop();
    }
    int top() {
        return q.front();
    }
    bool empty() {
        return q.empty();
    }
};
