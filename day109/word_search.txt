class Solution {
public:
    bool dfs(vector<vector<char>>& board, string& word,
             int i, int j, int idx) {
        // All characters matched
        if (idx == word.size())
            return true;
        // Out of bounds or character doesn't match
        if (i < 0 || i >= board.size() ||
            j < 0 || j >= board[0].size() ||
            board[i][j] != word[idx])
            return false;
        // Mark as visited
        char ch = board[i][j];
        board[i][j] = '#';
        // Explore 4 directions
        bool found =
            dfs(board, word, i + 1, j, idx + 1) ||
            dfs(board, word, i - 1, j, idx + 1) ||
            dfs(board, word, i, j + 1, idx + 1) ||
            dfs(board, word, i, j - 1, idx + 1);
        // Backtrack
        board[i][j] = ch;
        return found;
    }
    bool exist(vector<vector<char>>& board, string word) {
        int n = board.size();
        int m = board[0].size();
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (board[i][j] == word[0]) {
                    if (dfs(board, word, i, j, 0))
                        return true;
                }
            }
        }
        return false;
    }
};
