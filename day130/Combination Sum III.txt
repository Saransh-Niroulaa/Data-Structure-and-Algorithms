class Solution {
public:
    vector<vector<int>> ans;
    void solve(int start, int k, int target, vector<int>& ds) {
        if (k == 0) {
            if (target == 0)
                ans.push_back(ds);
            return;
        }
        for (int i = start; i <= 9; i++) {
            if (i > target)
                break;
            ds.push_back(i);
            solve(i + 1, k - 1, target - i, ds);
            ds.pop_back();
        }
    }
    vector<vector<int>> combinationSum3(int k, int n) {
        vector<int> ds;
        solve(1, k, n, ds);
        return ans;
    }
};
