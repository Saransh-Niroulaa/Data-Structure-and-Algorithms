#include <bits/stdc++.h>
using namespace std;

class Solution {
public:
    string postfixToPrefix(string s) {
        stack<string> st;

        for (char c : s) {
            if (isalnum(c)) {
                st.push(string(1, c));
            }
            else {
                string b = st.top();
                st.pop();

                string a = st.top();
                st.pop();

                st.push(string(1, c) + a + b);
            }
        }

        return st.top();
    }
};
