#include <bits/stdc++.h>
using namespace std;
class Solution {
public:
    string prefixToPostfix(string s) {
        stack<string> st;
        for (int i = s.size() - 1; i >= 0; i--) {
            char c = s[i];
            if (isalnum(c)) {
                st.push(string(1, c));
            }
            else {
                string a = st.top();
                st.pop();
                string b = st.top();
                st.pop();
                st.push(a + b + c);
            }
        }
        return st.top();
    }
};
