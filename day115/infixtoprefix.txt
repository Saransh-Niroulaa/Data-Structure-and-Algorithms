#include <bits/stdc++.h>
using namespace std;

class Solution {
public:
    int precedence(char c) {
        if (c == '+' || c == '-') return 1;
        if (c == '*' || c == '/') return 2;
        if (c == '^') return 3;
        return 0;
    }
    string infixToPrefix(string s) {
        reverse(s.begin(), s.end());
        for (char &c : s) {
            if (c == '(') c = ')';
            else if (c == ')') c = '(';
        }
        stack<char> st;
        string ans;
        for (char c : s) {
            if (isalnum(c))
                ans += c;
            else if (c == '(')
                st.push(c);
            else if (c == ')') {
                while (st.top() != '(') {
                    ans += st.top();
                    st.pop();
                }
                st.pop();
            }
            else {
                while (!st.empty() &&
                       precedence(c) < precedence(st.top())) {
                    ans += st.top();
                    st.pop();
                }
                st.push(c);
            }
        }
        while (!st.empty()) {
            ans += st.top();
            st.pop();
        }
        reverse(ans.begin(), ans.end());
        return ans;
    }
};
