#include <bits/stdc++.h>
using namespace std;

class Solution {
public:
    int precedence(char op) {
        if (op == '+' || op == '-') return 1;
        if (op == '*' || op == '/') return 2;
        return 0;
    }

    string infixToPostfix(string infix) {
        stack<char> st;
        string postfix = "";

        for (char c : infix) {
            if (isalnum(c))
                postfix += c;

            else if (c == '(')
                st.push(c);

            else if (c == ')') {
                while (st.top() != '(') {
                    postfix += st.top();
                    st.pop();
                }
                st.pop();
            }

            else {
                while (!st.empty() &&
                       precedence(c) <= precedence(st.top())) {
                    postfix += st.top();
                    st.pop();
                }
                st.push(c);
            }
        }

        while (!st.empty()) {
            postfix += st.top();
            st.pop();
        }

        return postfix;
    }
};
