class Solution {
public:
    int largestRectangleArea(vector<int>& a) {
        stack<int> st;
        int ans = 0, n = a.size();
        for (int i = 0; i <= n; i++) {
            int h = i == n ? 0 : a[i];
            while (!st.empty() && a[st.top()] > h) {
                int height = a[st.top()];
                st.pop();
                int width = st.empty() ? i : i - st.top() - 1;
                ans = max(ans, height * width);
            }
            st.push(i);
        }
        return ans;
    }

    int maximalRectangle(vector<vector<char>>& matrix) {
        if (matrix.empty()) return 0;
        int m = matrix[0].size(), ans = 0;
        vector<int> h(m);
        for (auto& row : matrix) {
            for (int j = 0; j < m; j++)
                h[j] = row[j] == '1' ? h[j] + 1 : 0;
            ans = max(ans, largestRectangleArea(h));
        }
        return ans;
    }
};
