class Solution {
public:
  long long mod=1e9+7;
   long long pow(long long a,long long b){
    long long res=1;
     while(b>0){
        if(b%2==1){
            res=(res*a)%mod;
        }
        a=(a*a)%mod;
        b/=2;
     }
     return res;
   }
    int countGoodNumbers(long long n) {
        long long  even=(n+1)/2;
        long long odd=n/2;
       return (pow(5,even)*pow(4,odd))%mod;
    }
};
