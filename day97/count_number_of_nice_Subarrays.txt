class Solution {
public:
    int atMost(vector<int>& nums, int k) {
        int l = 0;
        int count = 0;
        int ans = 0;
        for (int r = 0; r < nums.size(); r++) {
            if (nums[r] % 2)
                count++;
            while (count > k) {
                if (nums[l] % 2)
                    count--;
                l++;
            }
            ans += r - l + 1;
        }
        return ans;
    }
    int numberOfSubarrays(vector<int>& nums, int k) {
        return atMost(nums, k) - atMost(nums, k - 1);
    }
};
